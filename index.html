// Supabase Edge Function: send-sms // Sends one SMS via Africa's Talking. The API key never reaches the browser. // // Deploy via Supabase Dashboard -> Edge Functions -> Deploy a new function // -> name it "send-sms" -> paste this file's contents. // // Then set these secrets under Project Settings -> Edge Functions -> Secrets: // AT_USERNAME = your Africa's Talking username ("sandbox" while testing) // AT_API_KEY = your Africa's Talking API key // AT_ENV = "sandbox" or "production" // AT_SENDER_ID = your approved Sender ID (leave blank until you have one) import { serve } from "https://deno.land/std@0.168.0/http/server.ts"; const AT_USERNAME = Deno.env.get("AT_USERNAME") ?? ""; const AT_API_KEY = Deno.env.get("AT_API_KEY") ?? ""; const AT_ENV = Deno.env.get("AT_ENV") ?? "sandbox"; const AT_SENDER_ID = Deno.env.get("AT_SENDER_ID") ?? ""; const AT_URL = AT_ENV === "production" ? "https://api.africastalking.com/version1/messaging" : "https://api.sandbox.africastalking.com/version1/messaging"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; // Normalizes a Zambian number to +260XXXXXXXXX. Returns null if it doesn't look valid. function formatZambianPhone(raw) { if (!raw) return null; let p = String(raw).replace(/[^\d+]/g, ""); if (p.startsWith("+260") && p.length === 13) return p; if (p.startsWith("260") && p.length === 12) return "+" + p; if (p.startsWith("0") && p.length === 10) return "+260" + p.slice(1); if (p.length === 9) return "+260" + p; return null; } serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (!AT_USERNAME || !AT_API_KEY) { return new Response( JSON.stringify({ error: "SMS is not configured yet (missing AT_USERNAME / AT_API_KEY secrets)." }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } try { const { phone, message } = await req.json(); const formatted = formatZambianPhone(phone); if (!formatted) { return new Response( JSON.stringify({ error: `Invalid/unrecognized phone number: "${phone}"` }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } if (!message || !String(message).trim()) { return new Response( JSON.stringify({ error: "Message text is required." }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } const form = new URLSearchParams({ username: AT_USERNAME, to: formatted, message: String(message), }); if (AT_SENDER_ID) form.set("from", AT_SENDER_ID); const atRes = await fetch(AT_URL, { method: "POST", headers: { "apiKey": AT_API_KEY, "Content-Type": "application/x-www-form-urlencoded", "Accept": "application/json", }, body: form, }); const data = await atRes.json(); return new Response(JSON.stringify(data), { status: atRes.ok ? 200 : 502, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } catch (err) { return new Response( JSON.stringify({ error: String(err && err.message ? err.message : err) }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } });
